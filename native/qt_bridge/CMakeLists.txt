cmake_minimum_required(VERSION 3.21)
project(modhunter_qt_bridge VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(modhunter-qt-bridge SHARED qt_bridge.cpp)
target_link_libraries(modhunter-qt-bridge PRIVATE Qt6::Core)
target_compile_definitions(modhunter-qt-bridge PRIVATE QT_NO_KEYWORDS)
set_target_properties(modhunter-qt-bridge PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
  target_compile_options(modhunter-qt-bridge PRIVATE /W4 /WX)
else()
  target_compile_options(modhunter-qt-bridge PRIVATE -Wall -Wextra -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  add_library(modhunter-qt-bridge-fixture SHARED qt_bridge_fixture.cpp)
  target_link_libraries(modhunter-qt-bridge-fixture PRIVATE Qt6::Core)
  set_target_properties(modhunter-qt-bridge-fixture PROPERTIES PREFIX "")
  add_executable(modhunter-qt-bridge-test qt_bridge_test.cpp)
  target_link_libraries(modhunter-qt-bridge-test PRIVATE modhunter-qt-bridge Qt6::Core)
  add_test(NAME callback_sender_contract COMMAND modhunter-qt-bridge-test)
endif()
